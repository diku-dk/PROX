INCLUDE_DIRECTORIES(
  ${PROJECT_SOURCE_DIR}/PROX/FOUNDATION/GRID/GRID/include
  ${Boost_INCLUDE_DIRS}
)

FILE(
  GLOB
  GRID_SOURCES
  ${PROJECT_SOURCE_DIR}/PROX/FOUNDATION/GRID/GRID/src/*.cpp
  )

FILE(
  GLOB
  GRID_HEADERS
  ${PROJECT_SOURCE_DIR}/PROX/FOUNDATION/GRID/GRID/include/*.h
  )

ADD_DEFINITIONS("${3RDPARTY_DEF}")

ADD_LIBRARY(
  GRID
  ${GRID_SOURCES}
  ${GRID_HEADERS}
  )

TARGET_LINK_LIBRARIES( GRID )

SET_TARGET_PROPERTIES(GRID PROPERTIES DEBUG_POSTFIX "D" )

SET_TARGET_PROPERTIES(GRID PROPERTIES ARCHIVE_OUTPUT_DIRECTORY                "${PROJECT_SOURCE_DIR}/lib/")
SET_TARGET_PROPERTIES(GRID PROPERTIES ARCHIVE_OUTPUT_DIRECTORY_DEBUG          "${PROJECT_SOURCE_DIR}/lib/")
SET_TARGET_PROPERTIES(GRID PROPERTIES ARCHIVE_OUTPUT_DIRECTORY_RELEASE        "${PROJECT_SOURCE_DIR}/lib/")
SET_TARGET_PROPERTIES(GRID PROPERTIES ARCHIVE_OUTPUT_DIRECTORY_MINSIZEREL     "${PROJECT_SOURCE_DIR}/lib/")
SET_TARGET_PROPERTIES(GRID PROPERTIES ARCHIVE_OUTPUT_DIRECTORY_RELWITHDEBINFO "${PROJECT_SOURCE_DIR}/lib/")
